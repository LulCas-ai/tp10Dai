// materias-repository.js
import { pool } from '../db/pg-connection.js';

export async function getAllAsync() {
  const query = 'SELECT id, nombre, carga_horaria FROM materias ORDER BY id';
  const { rows } = await pool.query(query);
  return rows;
}

export async function getByIdAsync(id) {
  const query = 'SELECT id, nombre, carga_horaria FROM materias WHERE id = $1';
  const values = [id];
  const { rows } = await pool.query(query, values);
  return rows[0] || null;
}

export async function createAsync({ nombre, carga_horaria }) {
  const query = `
    INSERT INTO materias (nombre, carga_horaria)
    VALUES ($1, $2)
    RETURNING id, nombre, carga_horaria
  `;
  const values = [nombre, carga_horaria];
  const { rows } = await pool.query(query, values);
  return rows[0];
}

export async function updateAsync(id, { nombre, carga_horaria }) {
  const query = `
    UPDATE materias
    SET nombre = $1, carga_horaria = $2
    WHERE id = $3
    RETURNING id, nombre, carga_horaria
  `;
  const values = [nombre, carga_horaria, id];
  const { rows } = await pool.query(query, values);
  return rows[0] || null;
}

export async function deleteByIdAsync(id) {
  const query = 'DELETE FROM materias WHERE id = $1 RETURNING id';
  const values = [id];
  const { rows } = await pool.query(query, values);
  return rows[0] ? true : false;
}

Tenés toda la razón: el updateAsync que te pasé antes pisa los campos no enviados. Para resolverlo, hay que construir dinámicamente el SET en base a las propiedades definidas en el body, manteniendo placeholders para prevenir SQL Injection. Aquí te dejo una versión corregida y más robusta:

js
// materias-repository.js
import { pool } from '../db/pg-connection.js';

export async function updateAsync(id, { nombre, carga_horaria }) {
  const fields = [];
  const values = [];
  let index = 1;

  if (nombre !== undefined) {
    fields.push(`nombre = $${index}`);
    values.push(nombre);
    index++;
  }

  if (carga_horaria !== undefined) {
    fields.push(`carga_horaria = $${index}`);
    values.push(carga_horaria);
    index++;
  }

  if (fields.length === 0) {
    // No hay campos para actualizar
    return null;
  }

  // Agregamos el id al final
  values.push(id);

  const query = `
    UPDATE materias
    SET ${fields.join(', ')}
    WHERE id = $${index}
    RETURNING id, nombre, carga_horaria
  `;

  const { rows } = await pool.query(query, values);
  return rows[0] || null;
}
Detalles clave:
Se arma un array fields con los fragmentos columna = $n solo si el campo está definido.

Se mantiene el orden de los placeholders $1, $2, etc., en paralelo con el array values.

Si no hay campos para actualizar, devuelve null directamente.

El id siempre se agrega al final como último parámetro.

De esta forma, si el cliente manda solo { "carga_horaria": 5 }, la consulta resultante será:

sql
UPDATE materias SET carga_horaria = $1 WHERE id = $2 RETURNING id, nombre, carga_horaria;
Y no se pisa nombre.

js
// materias-controller.js
import express from 'express';
import { StatusCodes } from 'http-status-codes';
import * as MateriasService from '../services/materias-service.js';

const router = express.Router();

// GET /api/materias
router.get('/api/materias', async (req, res) => {
  try {
    const materias = await MateriasService.getAllAsync();
    res.status(StatusCodes.OK).json(materias);
  } catch (error) {
    res.status(StatusCodes.BAD_REQUEST).json({ error: error.message });
  }
});

// GET /api/materias/:id
router.get('/api/materias/:id', async (req, res) => {
  try {
    const materia = await MateriasService.getByIdAsync(req.params.id);
    if (!materia) {
      return res.status(StatusCodes.NOT_FOUND).json({ error: 'Materia no encontrada' });
    }
    res.status(StatusCodes.OK).json(materia);
  } catch (error) {
    res.status(StatusCodes.BAD_REQUEST).json({ error: error.message });
  }
});

// POST /api/materias
router.post('/api/materias', async (req, res) => {
  try {
    const nuevaMateria = await MateriasService.createAsync(req.body);
    res.status(StatusCodes.CREATED).json(nuevaMateria);
  } catch (error) {
    res.status(StatusCodes.BAD_REQUEST).json({ error: error.message });
  }
});

// PUT /api/materias/:id
router.put('/api/materias/:id', async (req, res) => {
  try {
    const materiaActualizada = await MateriasService.updateAsync(req.params.id, req.body);
    if (!materiaActualizada) {
      return res.status(StatusCodes.NOT_FOUND).json({ error: 'Materia no encontrada' });
    }
    res.status(StatusCodes.OK).json(materiaActualizada);
  } catch (error) {
    res.status(StatusCodes.BAD_REQUEST).json({ error: error.message });
  }
});

// DELETE /api/materias/:id
router.delete('/api/materias/:id', async (req, res) => {
  try {
    const eliminada = await MateriasService.deleteByIdAsync(req.params.id);
    if (!eliminada) {
      return res.status(StatusCodes.NOT_FOUND).json({ error: 'Materia no encontrada' });
    }
    res.status(StatusCodes.OK).json({ message: 'Materia eliminada correctamente' });
  } catch (error) {
    res.status(StatusCodes.BAD_REQUEST).json({ error: error.message });
  }
});

export default router;
Puntos clave:
Se usa express.Router() para definir las rutas.

Cada endpoint está envuelto en try/catch para capturar errores.

Respuestas con 200 OK, 201 Created, 404 Not Found y 400 Bad Request según corresponda.

Delegación directa a la capa de servicio (MateriasService).
